==============================
Rules for this codebase
------------------------------

- SS = DS.
- One data segment.
- One code segment, if I recall correctly?

==============================
Prologues and epilogues
------------------------------

Unused locals typically get removed, same deal with written-but-not-read locals.

Arguments start from [BP+0x4] for normal functions and [BP+0x6] for far functions.

No arguments or used locals:

  ...
  ret

1 argument, no used locals:

  push bp
  mov bp,sp
  ...
  pop bp
  ret

1 argument, 2 bytes of locals:

  push bp
  mov bp,sp
  sub sp,0x2
  ...
  leave
  ret

A `FAR PASCAL` Windows-callback-compatible function, 10 bytes of callee-cleanup args, 2 bytes of locals:

  mov ax,ds
  nop
  inc bp
  push bp
  mov bp,sp
  push ds
  mov ds,ax
  sub sp,0x2
  ...
  lea sp,[bp-0x2]
  pop ds
  pop bp
  dec bp
  retf word 0xa

==============================
Compiler intrinsics
------------------------------

*** strcpy(ds:dst, src="const string") ***

Constant string gets allocated onto DS.
NOTE: "length" includes the NUL terminator here.

It all seems to start like this:

  mov di,&dst
  mov si,src.ptr
  cld
  push ds
  pop es

Then:

If the length is divisible by 2:

  mov cx,src.len/2
  rep movsw

Otherwise, if the length mod 2 is 1:

  mov cx,floor(src.len/2)
  rep movsw
  movsb

Also, if the length is 1 (yeah, go figure with that one):

  xor cx,cx
  rep movsw
  movsb

*** strcpy(ds:dst, ds:src) ***

One case I have uses a `lea di,[bp-*]` to point to a destination buffer stored in locals.
But another case does use `mov di,&dst`.

  mov di,&dst
  push di
    mov di,&src
    cld
    push ds
    pop es
    mov ax,di
  pop di
  push ax
    xor ax,ax
    mov cx,0xffff
    repne scasb
    not cx
    sub di,cx
    shr cx,1
    push di
      push es
        push ds
        pop es
      pop ax
    pop si
  pop di
  push ds
    mov ds,ax
    rep movsw
    adc cx,cx
    rep movsb
  pop ds

*** strcat(ds:dst, src="const string") ***

Constant string gets allocated onto DS.
NOTE: "length" includes the NUL terminator here.

  mov di,&dst
  mov si,src.ptr
  cld
  mov cx,0xffff
  xor ax,ax
  repne scasb
  dec di
  mov cx,src.len
  rep movsb

*** strcat(ds:dst, ds:src) ***

  mov di,&dst
  push di
    mov di,&src
    cld
    xor ax,ax
    mov cx,0xffff
    repne scasb
    not cx
    sub di,cx
    push di
      push es
        push ds
        pop es
      pop ax
    pop si
  pop di
  mov word [bp-TMP_SEG],ds
  mov ds,ax
  push cx
    mov cx,0xffff
    xor ax,ax
    repne scasb
    dec di
  pop cx
  rep movsb
  mov ds,word [bp-TMP_SEG]

*** strchr(ds:src, const CHAR_TO_FIND) ***

  mov di,&src
  push ds
  pop es
  cld
  xor ax,ax
  mov cx,0xffff
  repne scasb
  not cx
  mov ax,CHAR_TO_FIND
  sub di,cx
  repne scasb
  jz @skip
    mov di,0x1
    xor ax,ax
    mov es,ax
  @skip:
  dec di

*** strlen(ds:src) -> AX ***

  mov di,&src
  push ds
  pop es
  cld
  xor ax,ax
  mov cx,0xffff
  repne scasb
  not cx
  dec cx
  mov ax,cx
